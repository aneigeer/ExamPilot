      startMockTimer: function () {
        this.stopMockTimer();
        var self = this;
        this.mockTimerId = setInterval(function () {
          if (self.mockRemainSec > 0) {
            self.mockRemainSec -= 1;

      stopMockTimer: function () {
        if (this.mockTimerId) {
          clearInterval(this.mockTimerId);
          this.mockTimerId = null;
        }

      confirmMockSubmit: function () {          // 手动交卷：二次确认
        if (this.mockSubmitting || !this.mockPaper) return;
        var total = this.mockPaper.questions ? this.mockPaper.questions.length : 0;
        var unanswered = Math.max(0, total - Object.keys(this.mockAnswers || {}).length);
        var msg = unanswered > 0
          ? ('还有 ' + unanswered + ' 题未作答，确定交卷吗？交卷后不可修改。')
          : '确定交卷吗？交卷后不可修改。';
        if (window.confirm(msg)) {
          this.doSelfMockSubmit();
        }
      },

      doSelfMockSubmit: function () {          // 保存当前题 → submit → report
        if (!this.mockPaper || this.mockSubmitting) return;

        }).then(function (report) {
          self.mockReport = report || {};
          self.mockPaper = null;
          self.saveMockRecordId(null);
          toast('已交卷');
        }).catch(function (e) {
          self.mockError = e.message || '交卷失败';
        }).then(function () {

      tryResumeMock: function () {
