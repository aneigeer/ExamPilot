## 1. 步骤与命令原文（按时间顺序）

```bash
# 1) 环境核对（只读）
mysql -uroot -p*** -N -e "SELECT VERSION();
  SELECT COUNT(*) FROM information_schema.tables WHERE table_schema='exam_prep';"
# → 8.0.46 / 56（全部 BASE TABLE，无视图）

# 2) 基线：逐表 SELECT COUNT(*) 生成 TSV（information_schema.table_rows 为估算值，不使用）
bash 项目规划/第9周/evidence/backup_drill_20261109_counts.sh exam_prep \
     项目规划/第9周/evidence/baseline_counts_20261109.tsv
# → tables=56，总行数 32869，用时 4.4s

# 3) 基线关键表校验摘要
mysql -uroot -p*** -N -B -e "SELECT COUNT(*),MIN(id),MAX(id) FROM exam_prep.ai_chunk;"
mysql -uroot -p*** -N -B -e "SELECT COUNT(*),MIN(id),MAX(id) FROM exam_prep.question;"
mysql -uroot -p*** -N -B -e "SELECT audit_status,COUNT(*) FROM exam_prep.question GROUP BY audit_status;"
mysql -uroot -p*** -N -B -e "SELECT COUNT(*),MIN(id),MAX(id) FROM exam_prep.\`user\`;"

# 4) 备份（含 routines/triggers，单事务一致性快照，utf8mb4）
time mysqldump --single-transaction --routines --triggers --default-character-set=utf8mb4 \
  -uroot -p*** exam_prep > 项目规划/第9周/evidence/exam_prep_backup_20261109.sql
# → exit=0，耗时 3 秒，24,891,258 字节

# 5) 备份产物校验
sha256sum 项目规划/第9周/evidence/exam_prep_backup_20261109.sql
grep -c "^CREATE TABLE" 项目规划/第9周/evidence/exam_prep_backup_20261109.sql
grep -c "^DROP TABLE IF EXISTS" 项目规划/第9周/evidence/exam_prep_backup_20261109.sql

# 6) 建隔离库并恢复
mysql -uroot -p*** -e "DROP DATABASE IF EXISTS \`exam_prep_drill_20261109\`;
  CREATE DATABASE \`exam_prep_drill_20261109\` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci;"
time mysql --default-character-set=utf8mb4 -uroot -p*** exam_prep_drill_20261109 \
  < 项目规划/第9周/evidence/exam_prep_backup_20261109.sql
# → exit=0，耗时 46 秒

# 7) 恢复库逐表行数 + 与基线逐表比对
bash 项目规划/第9周/evidence/backup_drill_20261109_counts.sh exam_prep_drill_20261109 \
     项目规划/第9周/evidence/restore_counts_20261109.tsv
diff 项目规划/第9周/evidence/baseline_counts_20261109.tsv \
     项目规划/第9周/evidence/restore_counts_20261109.tsv          # → 无差异
join -t $'\t' -j 1 <(sort 基线) <(sort 恢复后) | awk -F'\t' '$2!=$3{c++} END{print c+0}'
# → MISMATCH_COUNT=0

# 8) 抽样行级 MD5 比对（首行/中位行/末行，整行全列 CONCAT_WS 后 MD5）
mysql -uroot -p*** -N -B exam_prep                 < sample_check_20261109.sql
mysql -uroot -p*** -N -B exam_prep_drill_20261109  < sample_check_20261109.sql
mysql -uroot -p*** -N -B exam_prep                 < sample_check_20261109_norm.sql
mysql -uroot -p*** -N -B exam_prep_drill_20261109  < sample_check_20261109_norm.sql

# 9) 全表指纹 + 结构核对（引擎/索引/外键/自增值）
#    指纹: COUNT(*) 与 SUM(CRC32(CONCAT_WS('|', 全部列)))
#    索引: SELECT COUNT(DISTINCT table_name,index_name) FROM information_schema.statistics ...
```

> 注 1：`-p123456` 在命令中按团队给定的本机 root 口令执行，报告中以 `-p***` 脱敏。
> 注 2：Windows 版 `mysql -B` 输出为 CRLF，表名需 `tr -d '\r'`（首轮曾报 `ERROR 1103 Incorrect table name 'ai_agent_checkpoint'`，修正后通过）。
